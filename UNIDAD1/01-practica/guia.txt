05_operaciones
#!/bin/bash
read -p "Ingresa un numero (Presiona enter para 10)" num1
num1=${num1:-10}
read -p "Ingresa un numero (Presiona enter para 20)" num2
num2=${num2:-20}

suma=$((num1 + num2))
resta=$((num1 - num2))
multiplicacion=$((num1 * num2))
division=$((num1 / num2))
residuo=$((num1 % num2))
potencia=$((num1 ** num2))

echo "El resultado de la suma es: $suma"
echo "El resultado de la resta es: $resta"
echo "El resultado de la multiplicacion es: $multiplicacion"
echo "El resultado de la division es: $division"
echo "El resultado de el residuo es: $residuo"
echo "El resultado de la potencia es: $potencia"
